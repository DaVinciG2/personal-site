void mainImage(out vec4 fragColor, in vec2 fragCoord)
{
    vec2 uv = fragCoord / iResolution.xy;
    vec3 col = texture(iChannel0, uv).rgb;
    // Canvas spans document 100-200vh; WebGL y=0 is its bottom (200vh).
    // Spatial opacity remains attached to the page while scrolling.
    float distanceFrom200vh = uv.y;
#ifdef CLOUDS_ONLY
    float opacity = 1.0;
#else
    float opacity = 1.0 - smoothstep(0.0, 1.0, distanceFrom200vh);
#endif
    fragColor = vec4(col, opacity);
}
